* #### **Pasaje relacional.**

| VAR BASE RELATION Persona (     idPersona INTEGER,     nombre VARCHAR(30),     apellidos VARCHAR(30),     alias VARCHAR(30) NULLABLE,     fechaNacimiento DATE,     fechaComienzo DATE,     PRIMARY KEY {idPersona} ); VAR BASE RELATION Artista (     idArtista INTEGER,     idPersona INTEGER,     PRIMARY KEY {idArtista},     FOREIGN KEY (idPersona) REFERENCES Persona ); VAR BASE RELATION Autor (     idAutor INTEGER,     idPersona INTEGER,     PRIMARY KEY {idAutor},     FOREIGN KEY (idPersona) REFERENCES Persona ); VAR BASE RELATION Empresa (     idEmpresa INTEGER,     nombre VARCHAR(30),     PRIMARY KEY {idEmpresa} ); VAR BASE RELATION Comic (     idComic INTEGER,     nombre VARCHAR(50),     descripcion VARCHAR(255),     fechaInicio DATE,     popularidad FLOAT CHECK (popularidad \>= 0.0 AND popularidad \<= 5.0) ,     ejemplaresVendidos INTEGER,     idEmpresa INTEGER,     idAutor INTEGER,     PRIMARY KEY {idComic},     FOREIGN KEY (idEmpresa) REFERENCES Empresa,     FOREIGN KEY (idAutor) REFERENCES Autor ); VAR BASE RELATION Run (     idRun INTEGER,     fechaComienzo DATE,     popularidad FLOAT CHECK (popularidad \>= 0.0 AND popularidad \<= 5.0) ,     ejemplaresVendidos INTEGER,     idComic INTEGER,     PRIMARY KEY {idRun},     FOREIGN KEY (idComic) REFERENCES Comic ); VAR BASE RELATION Issue ( 	idIssue INTEGER, 	numero INTEGER, 	popularidad FLOAT CHECK (popularidad \>= 0.0 AND popularidad \<= 5.0) , 	ejemplaresVendidos INTEGER, 	idRun INTEGER, 	PRIMARY KEY {idIssue}, 	FOREIGN KEY (idRun) REFERENCES Run ); VAR BASE RELATION Cover ( 	idCover INTEGER, 	nombre VARCHAR(50), 	fechaCreacion DATE, 	esPrincipal BOOLEAN, 	idIssue INTEGER, 	idArtista INTEGER, 	PRIMARY KEY {idCover}, 	FOREIGN KEY (idIssue) REFERENCES Issue, 	FOREIGN KEY (idArtista) REFERENCES Artista ); VAR BASE RELATION ParticipacionArtista ( 	idParticipacionArtista INTEGER, 	rol VARCHAR(30), 	paginasAsignadas INTEGER, 	idIssue INTEGER, 	idArtista INTEGER, 	PRIMARY KEY {idParticipacionArtista}, 	FOREIGN KEY (idIssue) REFERENCES Issue, 	FOREIGN KEY (idArtista) REFERENCES Artista ); VAR BASE RELATION RunAutor ( 	idRunAutor INTEGER, 	rol VARCHAR(30), 	fechaInicio DATE, 	idRun INTEGER, 	idAutor INTEGER, 	PRIMARY KEY {idRunAutor}, 	FOREIGN KEY (idRun) REFERENCES Run, 	FOREIGN KEY (idAutor) REFERENCES Autor );
